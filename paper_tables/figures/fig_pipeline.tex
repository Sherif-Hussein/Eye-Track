\begin{tikzpicture}[

  node distance=9mm and 0.04\linewidth,

              minimum height=15mm, inner sep=3pt, line width=0.7pt,

\node[src] (s1) {\textbf{1\enspace Recording}\\[1pt] research tracker, or simulated webcam};
\node[src, right=of s1] (s2) {\textbf{2\enspace Gaze samples}\\[1pt] position on the screen, 10--1000~Hz};
\node[box, right=of s2] (s3) {\textbf{3\enspace Preprocessing}\\[1pt] anti-aliased resampling (simulated low rates and noise enter here); blinks marked or filled};
\node[box, right=of s3] (s4) {\textbf{4\enspace Looking on each item}\\[1pt] dwell directly from samples, or from detected fixations};
\node[ours, below=12mm of s1] (s5) {\textbf{5\enspace Features}\\[1pt] dwell over time, visits, first look; scaled within each person};
\node[ours, right=of s5] (s6) {\textbf{6\enspace Model}\\[1pt] small L2 logistic regression};
\node[ours, right=of s6] (s7) {\textbf{7\enspace Person score}\\[1pt] combines the item scores; threshold set on other people};
\node[box, right=of s7] (s8) {\textbf{8\enspace Output}\\[1pt] a recognition score, with its error rates};
